从upstream merge完验证命令

# 0. 先看本地feat/enterprise-platform分支是否包含最新tag的内容? 如果是"未包含"才往下进行
git fetch upstream --tags
git merge-base --is-ancestor dsh-v0.2.0-rc.2 feat/enterprise-platform; if ($?) { "已包含" } else { "未包含" }

# 1. 切到 master，同步官方最新
git checkout master
git fetch upstream --tags
git reset --hard dsh-v0.2.0-rc.2  # 推进到最新稳定 tag（替换成实际 tag 名）

# 2. 切到开发分支，合并 master
git checkout feat/enterprise-platform
git merge master

# 3. 如有冲突：
#    pnpm-lock.yaml 直接用官方版:
#      git checkout --theirs pnpm-lock.yaml
#      git add pnpm-lock.yaml
#    其他共享文件以官方为准，重新套企业改动
#    全部解决后: git add . && git merge --continue

# 4. 验证
pnpm install
pnpm run lint
pnpm run hygiene
pnpm run doc-sync
pnpm run typecheck
pnpm run build
pnpm dsh --profile enterprise

# 5. 只推 feat 到你的 fork
git push origin feat/enterprise-platform:master
